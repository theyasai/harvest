<!DOCTYPE html>

<html lang="ja">

<head>

  <meta charset="UTF-8">

  <meta name="viewport" content="width=device-width,initial-scale=1">

  <title>収穫記録</title>

</head>

<body style="font-family:sans-serif;text-align:center;padding:2rem">


  <h2 id="status">作業者カードをタッチ</h2>

  <p id="worker-name" style="font-size:1.4rem;font-weight:bold;color:#1a73e8"></p>

  <p id="msg" style="color:#555"></p>


  <script>

    // ★ここを書き換えてください

const GAS_URL = 'https://script.google.com/macros/s/AKfycbyLxPDiPKWvYxxXo_P4ibnbl3oPK_f8g7qHQUQfPDxnideCc3L19mDUtHw_jkWVqigf/exec';

    const TERMINAL   = localStorage.getItem('terminal') || '端末??';


    let currentWorker = localStorage.getItem('worker') || '';

    if (currentWorker) {

      document.getElementById('worker-name').textContent = currentWorker;

      document.getElementById('status').textContent = 'コンテナをタッチ';

    }


    // NFCタグをタッチした時（URLにパラメータが付く）

    const params = new URLSearchParams(location.search);

    const w   = params.get('worker');

    const uid = params.get('uid');


    if (w) {

      // 作業者カードをタッチ

      currentWorker = w;

      localStorage.setItem('worker', w);

      document.getElementById('worker-name').textContent = w;

      document.getElementById('status').textContent = 'コンテナをタッチ';

      document.getElementById('msg').textContent = w + ' さん、登録完了';

      history.replaceState({}, '', location.pathname);

    }


    if (uid && currentWorker) {

      // コンテナをタッチ → 記録送信

      document.getElementById('status').textContent = '送信中...';

      const url = `${GAS_URL}?worker=${encodeURIComponent(currentWorker)}`

        + `&uid=${encodeURIComponent(uid)}`

        + `&terminal=${encodeURIComponent(TERMINAL)}`;

      fetch(url)

        .then(() => {

          document.getElementById('status').textContent = '✅ 記録完了';

          document.getElementById('msg').textContent

            = currentWorker + ' / ' + uid;

          setTimeout(() => {

            document.getElementById('status').textContent = 'コンテナをタッチ';

            document.getElementById('msg').textContent = '';

          }, 2000);

        });

      history.replaceState({}, '', location.pathname);

    }


    // 端末セットアップ

    const setup = params.get('setup');

    if (setup) {

      localStorage.setItem('terminal', setup);

      document.getElementById('status').textContent

        = '端末番号をセット: ' + setup;

    }

  </script>

</body>

</html>
